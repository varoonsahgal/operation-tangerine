🐘 Aurora & the Data API

# Is the Data API on? Which ARN is the CLUSTER?
aws rds describe-db-clusters --db-cluster-identifier <id> --query 'DBClusters[0].[DBClusterArn,HttpEndpointEnabled]'

# Turn it on (Serverless v2 / provisioned)
aws rds enable-http-endpoint --resource-arn <cluster-arn>

# Run SQL
aws rds-data execute-statement --resource-arn <cluster-arn> --secret-arn <secret-arn> \
  --database agentdb --sql "SELECT 1" --format-records-as JSON

Agent access checklist: Data API on · rds-data:ExecuteStatement on the cluster ARN · secretsmanager:GetSecretValue on the secret · correct cluster ARN, secret ARN and database in config · DB user exists with GRANTs.

🧭 pgvector

CREATE EXTENSION IF NOT EXISTS vector;                       -- name is "vector"
SELECT extversion FROM pg_extension WHERE extname = 'vector';

-- column dims must equal the model's (Titan Text Embeddings V2 = 1024 by default)
SELECT format_type(atttypid, atttypmod) FROM pg_attribute
WHERE attrelid = 'bedrock_integration.bedrock_kb'::regclass AND attname = 'embedding';

-- operator ↔ opclass:  <=> cosine ↔ vector_cosine_ops · <-> L2 ↔ vector_l2_ops · <#> IP ↔ vector_ip_ops
CREATE INDEX ON t USING hnsw (embedding vector_cosine_ops) WITH (m = 16, ef_construction = 64);
SELECT id FROM t ORDER BY embedding <=> :q LIMIT 5;          -- needs ORDER BY … LIMIT to use the index
SET hnsw.ef_search = 100;                                     -- higher recall, slower

📚 Bedrock Knowledge Base on Aurora

CREATE SCHEMA IF NOT EXISTS bedrock_integration;
CREATE ROLE bedrock_user WITH LOGIN PASSWORD '…';             -- password = the KB secret's
GRANT USAGE ON SCHEMA bedrock_integration TO bedrock_user;
CREATE TABLE bedrock_integration.bedrock_kb (
  id uuid PRIMARY KEY, embedding vector(1024), chunks text, metadata json, custom_metadata jsonb);
GRANT SELECT, INSERT, UPDATE, DELETE ON bedrock_integration.bedrock_kb TO bedrock_user;
CREATE INDEX ON bedrock_integration.bedrock_kb USING hnsw (embedding vector_cosine_ops) WITH (ef_construction = 256);
CREATE INDEX ON bedrock_integration.bedrock_kb USING gin (to_tsvector('simple', chunks));
CREATE INDEX ON bedrock_integration.bedrock_kb USING gin (custom_metadata);

⚡ DynamoDB memory

aws dynamodb describe-table --table-name <t> --query 'Table.[KeySchema,GlobalSecondaryIndexes[].IndexName]'
aws dynamodb describe-time-to-live --table-name <t>
aws dynamodb query --table-name <t> --key-condition-expression "session_id = :s AND begins_with(sk, :p)" \
  --expression-attribute-values '{":s":{"S":"sess-1"},":p":{"S":"MSG#"}}' --no-scan-index-forward --limit 10
aws dynamodb get-item --table-name <t> --key '{"session_id":{"S":"sess-1"},"sk":{"S":"STATE"}}' --consistent-read

TTL = Number, epoch seconds. GSIs need table/NAME/index/* in IAM.

🏎️ Query optimization

EXPLAIN (ANALYZE, BUFFERS) SELECT … ;               -- look for Seq Scan + "Rows Removed by Filter"
CREATE EXTENSION IF NOT EXISTS pg_trgm;
CREATE INDEX ON products USING gin (name gin_trgm_ops);          -- ILIKE '%x%'
CREATE INDEX ON products (category, rating DESC, product_id);     -- WHERE category = ? ORDER BY rating DESC LIMIT n
ANALYZE products;
SELECT calls, mean_exec_time, left(query, 80) FROM pg_stat_statements ORDER BY total_exec_time DESC LIMIT 5;
